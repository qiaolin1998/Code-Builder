FUNCTION_BLOCK "状态控制程序"
VAR_INPUT
    // 输入信号
    I0_0 : BOOL;    // I0.0 输入
    I0_1 : BOOL;    // I0.1 输入
    I0_2 : BOOL;    // I0.2 输入
    I0_5 : BOOL;    // I0.5 输入
    I0_6 : BOOL;    // I0.6 输入
    I0_7 : BOOL;    // I0.7 输入
    I1_2 : BOOL;    // I1.2 输入
    I1_3 : BOOL;    // I1.3 输入
    I1_4 : BOOL;    // I1.4 输入
    I1_6 : BOOL;    // I1.6 输入
END_VAR

VAR_OUTPUT
    转阀 : BOOL;        // 转阀输出
    主阀 : BOOL;        // 主阀输出
    电机1 : BOOL;       // 电机1输出
    扭结电磁阀 : BOOL;  // 扭结电磁阀输出
END_VAR

VAR
    当前程序 : INT := 0;    // 程序状态：0=初始状态, 1-13=程序1-13
    程序7状态 : BOOL := FALSE;  // 程序7状态跟踪
    程序8状态 : BOOL := FALSE;  // 程序8状态跟踪
    
    电机1计时器 : TON;      // 电机1定时器
    电机1计时开始 : BOOL;   // 电机1计时启动标志
    电机1定时完成 : BOOL;   // 电机1定时完成标志
END_VAR

BEGIN
    // 电机1计时器，定时1秒
    电机1计时器(IN := 电机1计时开始,
               PT := T#1S,
               Q => 电机1定时完成);
    
    // 主状态机
    CASE 当前程序 OF
        0:  // 初始状态
            // 重置所有输出
            转阀 := FALSE;
            主阀 := FALSE;
            电机1 := FALSE;
            扭结电磁阀 := FALSE;
            
            // 重置程序7和程序8的状态
            程序7状态 := FALSE;
            程序8状态 := FALSE;
            
            // 重置电机1定时相关变量
            电机1计时开始 := FALSE;
            
            // 判断进入程序1还是程序2
            IF I0_7 THEN
                当前程序 := 1;
            ELSIF I1_6 THEN
                当前程序 := 2;
            END_IF;
            
        1:  // 程序1
            IF I0_5 THEN
                当前程序 := 3;
            END_IF;
            
        2:  // 程序2
            IF I0_0 AND I0_1 AND I0_6 THEN
                当前程序 := 4;
            END_IF;
            
        3:  // 程序3
            // 启动转阀
            转阀 := TRUE;
            
            IF I0_6 THEN
                当前程序 := 4;
            END_IF;
            
        4:  // 程序4
            // 启动主阀
            主阀 := TRUE;
            
            IF I0_2 THEN
                当前程序 := 5;
            END_IF;
            
        5:  // 程序5
            // 停止转阀
            转阀 := FALSE;
            当前程序 := 6;
            
        6:  // 程序6 - 同时进入程序7和程序8
            程序7状态 := TRUE;
            程序8状态 := TRUE;
            当前程序 := 7;  // 先处理程序7
            
        7:  // 程序7
            IF 程序7状态 THEN
                IF NOT I1_4 THEN  // I1.4断开
                    // 启动电机1
                    电机1 := TRUE;
                    
                    // 启动计时
                    IF NOT 电机1计时开始 THEN
                        电机1计时开始 := TRUE;
                    END_IF;
                    
                    // 检查计时是否完成
                    IF 电机1定时完成 THEN
                        电机1 := FALSE;  // 停止电机1
                        电机1计时开始 := FALSE;  // 重置计时器
                        程序7状态 := FALSE;  // 断开程序7
                    END_IF;
                ELSE  // I1.4导通
                    程序7状态 := FALSE;  // 断开程序7
                END_IF;
            END_IF;
            
            // 如果程序7已完成，检查程序8状态
            IF NOT 程序7状态 AND 程序8状态 THEN
                当前程序 := 8;
            ELSIF NOT 程序7状态 AND NOT 程序8状态 THEN
                // 如果程序7和程序8都完成，进入下一个状态
                当前程序 := 0;  // 返回初始状态
            END_IF;
            
        8:  // 程序8
            IF 程序8状态 THEN
                IF I0_5 THEN
                    当前程序 := 9;
                    程序8状态 := FALSE;
                END_IF;
            END_IF;
            
        9:  // 程序9
            // 停止主阀
            主阀 := FALSE;
            
            IF I0_1 THEN
                当前程序 := 10;
            END_IF;
            
        10: // 程序10
            IF NOT I1_4 THEN  // I1.4断开
                当前程序 := 0;  // 进入初始状态
            ELSIF I1_4 THEN  // I1.4导通
                当前程序 := 11;
            END_IF;
            
        11: // 程序11
            IF I1_2 THEN
                // 启动扭结电磁阀
                扭结电磁阀 := TRUE;
                当前程序 := 12;
            END_IF;
            
        12: // 程序12
            IF I1_3 THEN
                // 停止扭结电磁阀
                扭结电磁阀 := FALSE;
                当前程序 := 13;
            END_IF;
            
        13: // 程序13
            IF I1_2 THEN
                当前程序 := 0;  // 进入初始状态
            END_IF;
    END_CASE;
END_FUNCTION_BLOCK
